        async function addNewTaskOrVideo(e) {
            e.preventDefault();
            if(!supabaseClient) return;

                title: cleanInput(document.getElementById('t-title').value),
                link: cleanInput(document.getElementById('t-link').value),

        async function deleteTask(id) {
            if(!supabaseClient) return;
            const { error } = await supabaseClient.from('tasks').delete().eq('id', id);
            if(!error) {
                taskList = taskList.filter(t => t.id !== id);
                renderUI();
                triggerToast('টাস্ক ডিলিট করা হয়েছে!');
            } else {
                alert("ডিলিট করতে সমস্যা হয়েছে: " + error.message);
            }
        }

        async function toggleBlock(id) {
            let u = userList.find(x => x.id === id);
            if(u && supabaseClient) {
                let newStatus = u.status === 'Active' ? 'Blocked' : 'Active';
                const { error } = await supabaseClient.from('users').update({ status: newStatus }).eq('id', id);
                if(!error) { u.status = newStatus; renderUI(); triggerToast('স্ট্যাটাস পরিবর্তিত'); }
            }
        }

        async function modifyBalance(id) {
            let val = prompt("নতুন ব্যালেন্স লিখুন (টাকা):");
            if(val !== null && !isNaN(val) && supabaseClient) {
                let newBal = parseFloat(val);
                const { error } = await supabaseClient.from('users').update({ balance: newBal }).eq('id', id);
                if(!error) {
                    let u = userList.find(x => x.id === id);
                    if(u) u.balance = newBal;
                    renderUI();
                    triggerToast('ব্যালেন্স আপডেট হয়েছে!');
                }
            }
        }

                renderUI();
                triggerToast(`উইথড্র ${st} করা হয়েছে!`);
            }
        }

            `).join('');
            document.getElementById('users-table-body').innerHTML = userHTML || `<tr><td colspan="5" style="text-align:center; color:var(--text-muted);">কোনো ডাটা নেই</td></tr>`;

                if(w.status === 'Approved') totalPayout += (w.amount || 0);
